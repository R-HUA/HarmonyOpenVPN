cmake_minimum_required(VERSION 3.16)
project(OHOSOpenVPN LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(DEPS "${ROOT}/third_party")
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_PKGCONFIG_SUPPORT OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_CMAKE_PACKAGE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_OBJECT_LIBS OFF CACHE BOOL "" FORCE)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/mbedtls" mbedtls)
add_subdirectory("${DEPS}/jsoncpp" jsoncpp)
add_subdirectory("${DEPS}/fmt" fmt)
add_library(ovpn_lz4 STATIC "${DEPS}/lz4/lib/lz4.c")
target_include_directories(ovpn_lz4 PUBLIC "${DEPS}/lz4/lib")
set(OVPN "${CMAKE_CURRENT_SOURCE_DIR}/openvpn3")
# Keep pinned dependency sources intact; generate the corrected adapter in the build tree.
file(READ "${OVPN}/openvpn/mbedtls/crypto/cipher.hpp" CBC_ADAPTER)
set(CBC_ANCHOR "        initialized = true;")
string(FIND "${CBC_ADAPTER}" "${CBC_ANCHOR}" CBC_ANCHOR_POS)
if(CBC_ANCHOR_POS EQUAL -1)
  message(FATAL_ERROR "Pinned CBC adapter changed; review padding patch")
endif()
string(REPLACE "${CBC_ANCHOR}" "        if (mbedtls_cipher_get_cipher_mode(&ctx) == MBEDTLS_MODE_CBC &&\n            mbedtls_cipher_set_padding_mode(&ctx, MBEDTLS_PADDING_PKCS7) != 0)\n            throw mbedtls_cipher_error(\"mbedtls_cipher_set_padding_mode\");\n${CBC_ANCHOR}" CBC_ADAPTER "${CBC_ADAPTER}")
set(OVPN_COMPAT "${CMAKE_CURRENT_BINARY_DIR}/ovpn-compat")
file(MAKE_DIRECTORY "${OVPN_COMPAT}/openvpn/mbedtls/crypto")
file(WRITE "${OVPN_COMPAT}/openvpn/mbedtls/crypto/cipher.hpp" "${CBC_ADAPTER}")
add_library(vpn_client SHARED vpn_client.cpp "${OVPN}/client/ovpncli.cpp" "${OVPN}/openvpn/crypto/data_epoch.cpp")
target_sources(vpn_client PRIVATE "${ROOT}/tests/cbc_selftest.cpp")
target_compile_definitions(vpn_client PRIVATE OVPN_EMBEDDED_CBC_TEST)
target_include_directories(vpn_client BEFORE PRIVATE "${OVPN_COMPAT}")
target_include_directories(vpn_client PRIVATE "${OVPN}" "${DEPS}/asio/asio/include" "${DEPS}/jsoncpp/include")
target_compile_definitions(vpn_client PRIVATE USE_TUN_BUILDER USE_ASIO ASIO_STANDALONE USE_MBEDTLS HAVE_JSONCPP HAVE_LZ4)
target_link_libraries(vpn_client PRIVATE ovpn_lz4 mbedtls mbedx509 mbedcrypto jsoncpp_static fmt::fmt ace_napi.z hilog_ndk.z)
target_compile_options(vpn_client PRIVATE -include mbedtls/oid.h)
add_executable(cbc_selftest EXCLUDE_FROM_ALL "${ROOT}/tests/cbc_selftest.cpp")
target_include_directories(cbc_selftest BEFORE PRIVATE "${OVPN_COMPAT}" "${OVPN}" "${DEPS}/asio/asio/include")
target_compile_definitions(cbc_selftest PRIVATE USE_MBEDTLS USE_ASIO ASIO_STANDALONE)
target_link_libraries(cbc_selftest PRIVATE mbedcrypto fmt::fmt)
